export default function(canvas) {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  renderer.size(canvas.clientWidth, canvas.clientHeight, false);
  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0xfef3c7);
  const camera = new THREE.PerspectiveCamera(50, canvas.clientWidth / canvas.clientHeight, 0.1, 100);
  camera.position.set(0, 3, 8);

  scene.add(new THREE.AmbientLight(0xffffff, 0.6));
  const sun = new THREE.DirectionalLight(0xffffff, 0.9);
  sun.position.set(3, 5, 6);
  scene.add(sun);

  // Ground plane
  const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(14, 4),
    new THREE.MeshStandardMaterial({ color: 0xe0f2fe })
  );
  ground.rotation.x = -Math.PI / 2;
  ground.position.y = -0.01;
  scene.add(ground);

  // --- Fraction bar 1: 1/2 ---
  // Total bar (background) for 1/2
  const bar1Total = new THREE.Mesh(
    new THREE.BoxGeometry(3, 0.5, 0.15),
    new THREE.MeshStandardMaterial({ color: 0xf1f5f9 })
  );
  bar1Total.position.set(-3.5, 1.0, 0);
  scene.add(bar1Total);

  // Divider line for 1/2 (middle)
  const div1 = new THREE.Mesh(
    new THREE.BoxGeometry(0.04, 0.55, 0.18),
    new THREE.MeshStandardMaterial({ color: 0x475569 })
  );
  div1.position.set(-3.5, 1.0, 0);
  scene.add(div1);

  // Shaded portion for 1/2 (left half) — starts invisible, animates in
  const shade1 = new THREE.Mesh(
    new THREE.BoxGeometry(0, 0.46, 0.16),
    new THREE.MeshStandardMaterial({ color: 0x3b82f6 })
  );
  shade1.position.set(-4.25, 1.0, 0);
  scene.add(shade1);

  // Label "1/2"
  // We'll use small boxes to form text-like markers instead — but simpler: use a plane with canvas texture
  // Actually let's create number labels using small geometries
  // We'll skip text and use color-coded legend boxes instead

  // --- Fraction bar 2: 1/3 ---
  const bar2Total = new THREE.Mesh(
    new THREE.BoxGeometry(3, 0.5, 0.15),
    new THREE.MeshStandardMaterial({ color: 0xf1f5f9 })
  );
  bar2Total.position.set(-3.5, 0.0, 0);
  scene.add(bar2Total);

  // Two dividers for 1/3 (at 1/3 and 2/3)
  const div2a = new THREE.Mesh(
    new THREE.BoxGeometry(0.04, 0.55, 0.18),
    new THREE.MeshStandardMaterial({ color: 0x475569 })
  );
  div2a.position.set(-4.5, 0.0, 0);
  scene.add(div2a);

  const div2b = new THREE.Mesh(
    new THREE.BoxGeometry(0.04, 0.55, 0.18),
    new THREE.MeshStandardMaterial({ color: 0x475569 })
  );
  div2b.position.set(-3.5, 0.0, 0);
  scene.add(div2b);

  // Shaded portion for 1/3 (leftmost third) — animates in
  const shade2 = new THREE.Mesh(
    new THREE.BoxGeometry(0, 0.46, 0.16),
    new THREE.MeshStandardMaterial({ color: 0xef4444 })
  );
  shade2.position.set(-4.83, 0.0, 0);
  scene.add(shade2);

  // --- "+" sign ---
  const plusV = new THREE.Mesh(
    new THREE.BoxGeometry(0.08, 0.5, 0.15),
    new THREE.MeshStandardMaterial({ color: 0x1e293b })
  );
  plusV.position.set(-0.8, 0.5, 0);
  scene.add(plusV);
  const plusH = new THREE.Mesh(
    new THREE.BoxGeometry(0.5, 0.08, 0.15),
    new THREE.MeshStandardMaterial({ color: 0x1e293b })
  );
  plusH.position.set(-0.8, 0.5, 0);
  scene.add(plusH);

  // --- "=" sign ---
  const eq1 = new THREE.Mesh(
    new THREE.BoxGeometry(0.5, 0.08, 0.15),
    new THREE.MeshStandardMaterial({ color: 0x1e293b })
  );
  eq1.position.set(1.5, 0.55, 0);
  scene.add(eq1);
  const eq2 = new THREE.Mesh(
    new THREE.BoxGeometry(0.5, 0.08, 0.15),
    new THREE.MeshStandardMaterial({ color: 0x1e293b })
  );
  eq2.position.set(1.5, 0.35, 0);
  scene.add(eq2);

  // --- Result bar: 5/6 ---
  const bar3Total = new THREE.Mesh(
    new THREE.BoxGeometry(3, 0.5, 0.15),
    new THREE.MeshStandardMaterial({ color: 0xf1f5f9 })
  );
  bar3Total.position.set(3.8, 0.5, 0);
  scene.add(bar3Total);

  // 5 dividers for 1/6 segments
  for (let i = 0; i < 5; i++) {
    const d = new THREE.Mesh(
      new THREE.BoxGeometry(0.04, 0.55, 0.18),
      new THREE.MeshStandardMaterial({ color: 0x475569 })
    );
    d.position.set(2.633 + i * 0.5, 0.5, 0);
    scene.add(d);
  }

  // 5 shaded segments for 5/6 — animates in
  const resultShades = [];
  for (let i = 0; i < 5; i++) {
    const s = new THREE.Mesh(
      new THREE.BoxGeometry(0, 0.46, 0.16),
      new THREE.MeshStandardMaterial({ color: i < 3 ? 0x3b82f6 : 0xef4444 })
    );
    s.position.set(2.383 + i * 0.5, 0.5, 0);
    scene.add(s);
    resultShades.push(s);
  }

  // --- Legend ---
  // Blue = 1/2, Red = 1/3
  const legendBlue = new THREE.Mesh(
    new THREE.BoxGeometry(0.3, 0.3, 0.15),
    new THREE.MeshStandardMaterial({ color: 0x3b82f6 })
  );
  legendBlue.position.set(-5.5, 2.2, 0);
  scene.add(legendBlue);

  const legendRed = new THREE.Mesh(
    new THREE.BoxGeometry(0.3, 0.3, 0.15),
    new THREE.MeshStandardMaterial({ color: 0xef4444 })
  );
  legendRed.position.set(-5.5, 1.6, 0);
  scene.add(legendRed);

  // --- Title area: use small blocks to hint at "1/2 + 1/3 = 5/6" ---
  // We'll create a result highlight box that appears at the end
  const resultHighlight = new THREE.Mesh(
    new THREE.BoxGeometry(3.2, 0.7, 0.05),
    new THREE.MeshStandardMaterial({ color: 0xfbbf24, transparent: true, opacity: 0 })
  );
  resultHighlight.position.set(3.8, 0.5, -0.1);
  scene.add(resultHighlight);

  // --- Animation ---
  // Phase 1 (0-2s): show 1/2 bar shaded
  // Phase 2 (2-4s): show 1/3 bar shaded
  // Phase 3 (4-6s): show result bar with 5/6 shaded
  // Phase 4 (6-7s): hold with highlight
  // Then loop

  const totalDuration = 7;
  let stopped = false;
  let paused = false;
  const clock = new THREE.Clock();

  function easeOut(t) { return 1 - Math.pow(1 - t, 3); }

  function tick() {
    if (stopped) return;
    if (!paused) {
      let elapsed = clock.getElapsedTime() % totalDuration;

      // Phase 1: shade 1/2 (0 to 2s)
      if (elapsed < 2) {
        const k = easeOut(elapsed / 2);
        shade1.scale.x = k;
        shade1.position.x = -4.25;
        shade1.geometry.dispose();
        shade1.geometry = new THREE.BoxGeometry(1.5 * k, 0.46, 0.16);
        shade1.position.x = -4.25 + (1.5 * k) / 2;
      } else {
        shade1.geometry.dispose();
        shade1.geometry = new THREE.BoxGeometry(1.5, 0.46, 0.16);
        shade1.scale.x = 1;
        shade1.position.x = -4.25 + 0.75;
      }

      // Phase 2: shade 1/3 (2 to 4s)
      if (elapsed >= 2 && elapsed < 4) {
        const k = easeOut((elapsed - 2) / 2);
        shade2.geometry.dispose();
        shade2.geometry = new THREE.BoxGeometry(1.0 * k, 0.46, 0.16);
        shade2.scale.x = 1;
        shade2.position.x = -4.83 + (1.0 * k) / 2;
      } else if (elapsed >= 4) {
        shade2.geometry.dispose();
        shade2.geometry = new THREE.BoxGeometry(1.0, 0.46, 0.16);
        shade2.scale.x = 1;
        shade2.position.x = -4.83 + 0.5;
      } else {
        shade2.geometry.dispose();
        shade2.geometry = new THREE.BoxGeometry(0, 0.46, 0.16);
        shade2.position.x = -4.83;
      }

      // Phase 3: result 5/6 (4 to 6s)
      if (elapsed >= 4 && elapsed < 6) {
        const k = easeOut((elapsed - 4) / 2);
        const segments = Math.floor(k * 5);
        const frac = (k * 5) - segments;
        for (let i = 0; i < 5; i++) {
          if (i < segments) {
            resultShades[i].geometry.dispose();
            resultShades[i].geometry = new THREE.BoxGeometry(0.46, 0.46, 0.16);
            resultShades[i].scale.x = 1;
          } else if (i === segments) {
            resultShades[i].geometry.dispose();
            resultShades[i].geometry = new THREE.BoxGeometry(0.46 * frac, 0.46, 0.16);
            resultShades[i].scale.x = 1;
          } else {
            resultShades[i].geometry.dispose();
            resultShades[i].geometry = new THREE.BoxGeometry(0, 0.46, 0.16);
            resultShades[i].scale.x = 1;
          }
        }
      } else if (elapsed >= 6) {
        for (let i = 0; i < 5; i++) {
          resultShades[i].geometry.dispose();
          resultShades[i].geometry = new THREE.BoxGeometry(0.46, 0.46, 0.16);
          resultShades[i].scale.x = 1;
        }
      } else {
        for (let i = 0; i < 5; i++) {
          resultShades[i].geometry.dispose();
          resultShades[i].geometry = new THREE.BoxGeometry(0, 0.46, 0.16);
          resultShades[i].scale.x = 1;
        }
      }

      // Phase 4: highlight result
      if (elapsed >= 6) {
        resultHighlight.material.opacity = 0.5;
      } else {
        resultHighlight.material.opacity = 0;
      }
    }

    renderer.render(scene, camera);
    requestAnimationFrame(tick);
  }

  tick();

  return {
    stop: () => { stopped = true; renderer.dispose(); },
    setPaused: (p) => { paused = p; clock.getDelta(); }
  };
}
