export default function(canvas) {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  const w = canvas.clientWidth || 800;
  const h = canvas.clientHeight || 600;
  renderer.setSize(w, h, false);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2));

  const scene = new THREE.Scene();
  scene.background = new THREE.Color(0xfef3c7);

  const camera = new THREE.PerspectiveCamera(50, w / h, 0.1, 200);
  camera.position.set(0, 3, 12);
  camera.lookAt(0, 0, 0);

  scene.add(new THREE.AmbientLight(0xffffff, 0.55));
  const sun = new THREE.DirectionalLight(0xffffff, 0.9);
  sun.position.set(5, 8, 10);
  scene.add(sun);

  // Ground
  const ground = new THREE.Mesh(
    new THREE.PlaneGeometry(30, 4),
    new THREE.MeshStandardMaterial({ color: 0xd1fae5 })
  );
  ground.rotation.x = -Math.PI / 2;
  ground.position.y = -0.25;
  scene.add(ground);

  // Road
  const road = new THREE.Mesh(
    new THREE.PlaneGeometry(20, 0.6),
    new THREE.MeshStandardMaterial({ color: 0x94a3b8 })
  );
  road.rotation.x = -Math.PI / 2;
  road.position.y = -0.2;
  scene.add(road);

  // Markers A and B
  const markerGeo = new THREE.CylinderGeometry(0.08, 0.08, 1.2, 16);
  const markerA = new THREE.Mesh(markerGeo, new THREE.MeshStandardMaterial({ color: 0x3b82f6 }));
  markerA.position.set(-8, 0.6, 0);
  scene.add(markerA);

  const markerB = new THREE.Mesh(markerGeo, new THREE.MeshStandardMaterial({ color: 0xef4444 }));
  markerB.position.set(8, 0.6, 0);
  scene.add(markerB);

  // Labels (small boxes as visual stand-ins)
  const labelGeo = new THREE.BoxGeometry(1.2, 0.5, 0.1);
  const labelA = new THREE.Mesh(labelGeo, new THREE.MeshStandardMaterial({ color: 0x3b82f6 }));
  labelA.position.set(-8, 1.6, 0);
  scene.add(labelA);

  const labelB = new THREE.Mesh(labelGeo, new THREE.MeshStandardMaterial({ color: 0xef4444 }));
  labelB.position.set(8, 1.6, 0);
  scene.add(labelB);

  // Distance label bar (shows 100km)
  const distBar = new THREE.Mesh(
    new THREE.BoxGeometry(16, 0.12, 0.12),
    new THREE.MeshStandardMaterial({ color: 0x6b7280 })
  );
  distBar.position.set(0, 2.3, 0);
  scene.add(distBar);

  // Xiaoming (blue sphere) starts at A (left, x=-8)
  const xiaoming = new THREE.Mesh(
    new THREE.SphereGeometry(0.35, 32, 16),
    new THREE.MeshStandardMaterial({ color: 0x3b82f6 })
  );
  xiaoming.position.set(-8, 0.35, 0.5);
  scene.add(xiaoming);

  // Xiaohua (red sphere) starts at B (right, x=8)
  const xiaohua = new THREE.Mesh(
    new THREE.SphereGeometry(0.35, 32, 16),
    new THREE.MeshStandardMaterial({ color: 0xef4444 })
  );
  xiaohua.position.set(8, 0.35, 0.5);
  scene.add(xiaohua);

  // Trail markers (small dots showing path)
  const trailGeo = new THREE.SphereGeometry(0.06, 8, 8);
  const xiaomingTrailMat = new THREE.MeshStandardMaterial({ color: 0x93c5fd });
  const xiaohuaTrailMat = new THREE.MeshStandardMaterial({ color: 0xfca5a5 });

  // Meeting point indicator (star-like)
  const meetMarker = new THREE.Mesh(
    new THREE.SphereGeometry(0.2, 16, 16),
    new THREE.MeshStandardMaterial({ color: 0xfacc15 })
  );
  meetMarker.visible = false;
  scene.add(meetMarker);

  // Problem parameters
  const totalDist = 100; // km
  const vMing = 4; // km/h
  const vHua = 6; // km/h
  const meetTime = totalDist / (vMing + vHua); // 10 hours
  const meetRatio = vMing / (vMing + vHua); // 0.4 from A

  // Animation duration in seconds (scaled for visual)
  const animDuration = 8;
  const holdDuration = 1.5;
  const totalLoop = animDuration + holdDuration;

  // Xiaoming travels 4/10 of the way = 40km; Xiaohua = 60km
  // On our 16-unit road: xiaoming moves from -8 to -8 + 16*0.4 = -1.6
  // xiaohua moves from 8 to 8 - 16*0.6 = -1.6
  const meetX = -8 + 16 * meetRatio; // -1.6

  let stopped = false;
  let paused = false;
  const clock = new THREE.Clock();

  function tick() {
    if (stopped) return;
    if (!paused) {
      let elapsed = clock.getElapsedTime();
      // Loop
      elapsed = elapsed % totalLoop;

      if (elapsed <= animDuration) {
        const k = elapsed / animDuration;
        // Ease slightly for readability
        const ek = k * k * (3 - 2 * k); // smoothstep

        xiaoming.position.x = -8 + (meetX - (-8)) * ek;
        xiaohua.position.x = 8 + (meetX - 8) * ek;

        // Bobbing
        xiaoming.position.y = 0.35 + Math.sin(elapsed * 6) * 0.08;
        xiaohua.position.y = 0.35 + Math.sin(elapsed * 6 + Math.PI) * 0.08;

        meetMarker.visible = false;
      } else {
        // Hold at meeting point
        xiaoming.position.x = meetX;
        xiaohua.position.x = meetX;
        xiaoming.position.y = 0.35;
        xiaohua.position.y = 0.35;
        meetMarker.position.set(meetX, 0.8, 0.5);
        meetMarker.visible = true;
      }
    }
    renderer.render(scene, camera);
    requestAnimationFrame(tick);
  }
  tick();

  return {
    stop: () => { stopped = true; renderer.dispose(); },
    setPaused: (p) => { paused = p; clock.getDelta(); }
  };
}
